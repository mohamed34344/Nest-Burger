abstract class Vehicle {

    double fuel;

    Vehicle(double fuel) {
        this.fuel = fuel;
    }

    void checkFuel() {
        if (fuel > 0) {
            System.out.println("Fuel is available");
        } else {
            System.out.println("No fuel");
        }
    }
}

class Car extends Vehicle {

    Car(double fuel) {
        super(fuel);
    }

    void drive() {
        System.out.println("Car is driving");
    }
}

class Motorcycle extends Vehicle {

    Motorcycle(double fuel) {
        super(fuel);
    }

    void ride() {
        System.out.println("Motorcycle is riding");
    }
}

public class Main {
    public static void main(String[] args) {

        Car c1 = new Car(20);
        c1.checkFuel();
        c1.drive();

        Motorcycle m1 = new Motorcycle(0);
        m1.checkFuel();
        m1.ride();
    }
}
